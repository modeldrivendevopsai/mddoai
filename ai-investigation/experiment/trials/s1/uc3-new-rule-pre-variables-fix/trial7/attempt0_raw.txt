```yaml
stages:
  - build
  - unitTest
  - healthCheck
  - secretScan
  - push

variables:
  IMAGE_NAME: "${CI_REGISTRY_IMAGE}/framework-chatbot-image"
  IMAGE_TAG: "${IMAGE_NAME}:${IMAGE_VERSION}-${CI_COMMIT_REF_SLUG}"

workflow:
  rules:
    - if: $CI_PIPELINE_SOURCE == "parent_pipeline"
    - if: $CI_COMMIT_BRANCH

default:
  image: docker:latest
  services:
    - docker:dind

build:
  stage: build
  tags:
    - ai-shared
  script:
    - docker build -t ${IMAGE_TAG} .

unitTest:
  stage: unitTest
  image: python:3.12-slim
  tags:
    - ai-shared
  before_script:
    - python3 -m pip install --break-system-packages -r requirements.txt
  script:
    - python3 test/run_unit_tests.py

healthCheck:
  stage: healthCheck
  image: curlimages/curl:latest
  tags:
    - ai-shared
  needs:
    - job: build
      artifacts: true
  script:
    - ./scripts/health_check.sh

secretScan:
  stage: secretScan
  tags:
    - ai-shared
  needs:
    - job: healthCheck
      artifacts: true
  script:
    - gitleaks detect --source . --no-git -v

push:
  stage: push
  tags:
    - ai-shared
  needs:
    - job: secretScan
      artifacts: true
  script:
    - docker login -u ${CI_REGISTRY_USER} -p ${CI_JOB_TOKEN} ${CI_REGISTRY}
    - docker push ${IMAGE_TAG}
```
