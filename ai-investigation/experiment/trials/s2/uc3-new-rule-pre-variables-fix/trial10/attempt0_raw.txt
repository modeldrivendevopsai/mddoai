```yaml
stages:
  - build
  - test
  - health_check
  - secret_scan
  - push

variables:
  IMAGE_NAME: chatbot-app

build:
  stage: build
  image: docker:latest
  services:
    - docker:dind
  script:
    - docker build -t $IMAGE_NAME .

test:
  stage: test
  image: python:3.11
  needs:
    - build
  script:
    - python test/run_unit_tests.py

health_check:
  stage: health_check
  image: docker:latest
  services:
    - docker:dind
  needs:
    - test
  script:
    - ./scripts/health_check.sh

secret_scan:
  stage: secret_scan
  image: zricethezav/gitleaks:latest
  needs:
    - health_check
  script:
    - gitleaks detect --source . --no-git -v --exit-code 1

push:
  stage: push
  image: docker:latest
  services:
    - docker:dind
  needs:
    - secret_scan
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD" $CI_REGISTRY
    - docker push $IMAGE_NAME
```
